Функция ОбработатьEvent(Запрос) Экспорт
	Попытка
		Парам = Запрос.ПараметрыЗапроса;
		Действие = Строка(Парам.Получить("action"));
		DealID   = Строка(Парам.Получить("deal_id"));
		Секрет   = Строка(Парам.Получить("secret"));

		ОжидаемыйСекрет = СокрЛП(Строка(Константы.B24_Секрет.Получить()));
		Если ЗначениеЗаполнено(ОжидаемыйСекрет) И СокрЛП(Секрет) <> ОжидаемыйСекрет Тогда
			Возврат СформироватьОтвет(401, "Unauthorized");
		КонецЕсли;

		Если ПустаяСтрока(DealID) Тогда
			Возврат СформироватьОтвет(400, "Bad request: deal_id required");
		КонецЕсли;

		// 1. Фиксируем пакет
		Пакет = Документы.B24_ВходящийПакет.СоздатьДокумент();
		Пакет.Дата = ТекущаяДата();
		Пакет.BitrixID = DealID;
		Пакет.ТипСобытия = Действие;
		Пакет.Статус = Перечисления.B24_СтатусПакета.Новый;
		Пакет.Записать();

		// 2. Сразу обрабатываем
		ОбработатьПакет(Пакет.Ссылка);

		Возврат СформироватьОтвет(200, "{""status"":""ok"",""id"":""" + Пакет.Номер + """}");
	Исключение
		Возврат СформироватьОтвет(500, "Internal error: " + ОписаниеОшибки());
	КонецПопытки;
КонецФункции

Процедура ОбработатьПакет(СсылкаПакета) Экспорт
	ОбъектПакета = СсылкаПакета.ПолучитьОбъект();
	Попытка
		// Идем в Битрикс за данными
		ДанныеИзБ24 = ПолучитьДанныеСделкиИзБ24(ОбъектПакета.BitrixID);
		
		// Сохраняем JSON для истории (как у тебя и было)
		ЗаписьJSON = Новый ЗаписьJSON;
		ЗаписьJSON.УстановитьСтроку();
		ЗаписатьJSON(ЗаписьJSON, ДанныеИзБ24);
		ОбъектПакета.JSONДанные = ЗаписьJSON.Закрыть();
		
		СозданнаяСсылка = Неопределено;
		
		// Определяем логику создания документов
		Если ОбъектПакета.ТипСобытия = "invoice" Тогда
			СозданнаяСсылка = СоздатьСчетНаОплату(ДанныеИзБ24);
		ИначеЕсли ОбъектПакета.ТипСобытия = "realization" Тогда
			СозданнаяСсылка = СоздатьРеализацию(ДанныеИзБ24);
		ИначеЕсли ОбъектПакета.ТипСобытия = "payment" Тогда
			СозданнаяСсылка = СоздатьПоступлениеНаСчет(ДанныеИзБ24);
		КонецЕсли;

		Если ЗначениеЗаполнено(СозданнаяСсылка) Тогда
			ОбъектПакета.СозданныйДокумент = СозданнаяСсылка;
			ОбъектПакета.Статус = Перечисления.B24_СтатусПакета.Обработан;
			
			// --- НОВОЕ: Отправка на почту, если e-mail пришел из Б24 ---
			Если ДанныеИзБ24.Свойство("client_email") И ЗначениеЗаполнено(ДанныеИзБ24.client_email) Тогда
				ОтправитьДокументНаПочту(СозданнаяСсылка, ДанныеИзБ24.client_email);
			КонецЕсли;
			
		Иначе
			ОбъектПакета.Статус = Перечисления.B24_СтатусПакета.Ошибка;
			ОбъектПакета.СообщениеОшибки = "Документ не создан по неизвестной причине.";
		КонецЕсли;
		
	Исключение
		ОбъектПакета.Статус = Перечисления.B24_СтатусПакета.Ошибка;
		ОбъектПакета.СообщениеОшибки = ОписаниеОшибки();
		// Пишем в системный журнал 1С
		ЗаписьЖурналаРегистрации("B24_Integration", УровеньЖурналаРегистрации.Ошибка, , СсылкаПакета, ОписаниеОшибки());
	КонецПопытки;
	ОбъектПакета.Записать();
КонецПроцедуры

Функция СоздатьСчетНаОплату(ДанныеJson) Экспорт
	Организация = ПолучитьОрганизациюПоУмолчанию();
	Контрагент  = НайтиИлиСоздатьКонтрагента(ДанныеJson.client_name);
	Договор     = НайтиИлиСоздатьДоговор(Контрагент);

	Док = Документы.СчетНаОплатуПокупателю.СоздатьДокумент();
	
	// 1. ПЕРВИЧНОЕ ЗАПОЛНЕНИЕ
	// взять основной счет, ответственных и другие настройки.
	Док.Заполнить(Организация); 
	
	// 2. УСТАНОВКА ОСТАЛЬНЫХ ДАННЫХ
	Док.Дата = ТекущаяДата();
	Док.Организация = Организация;
	Док.Контрагент = Контрагент;
	Док.ДоговорКонтрагента = Договор;
	
	// На всякий случай дублируем основной счет явно
	БанкДляСчета = Организация.ОсновнойБанковскийСчет;
	Попытка Док.БанковскийСчет = БанкДляСчета; Исключение КонецПопытки;
	Попытка Док.БанковскийСчетОрганизации = БанкДляСчета; Исключение КонецПопытки;

	Док.СуммаВключаетНДС = Истина; 
	ТекСтавка = Перечисления.СтавкиНДС.НДС20;

	// 3. ЗАПОЛНЕНИЕ ТОВАРОВ
	Для Каждого ТекТовар Из ДанныеJson.products Цикл
		НовСтр = Док.Товары.Добавить();
		НовСтр.Номенклатура = НайтиИлиСоздатьНоменклатуру(ТекТовар.name);
		НовСтр.Количество   = ТекТовар.quantity;
		НовСтр.Цена         = ТекТовар.price;
		НовСтр.СтавкаНДС    = ТекСтавка;
		НовСтр.Сумма        = Число(НовСтр.Количество) * Число(НовСтр.Цена);
		НовСтр.СуммаНДС     = Окр(НовСтр.Сумма * 20 / 120, 2);
	КонецЦикла;

	Док.СуммаДокумента = Док.Товары.Итог("Сумма");

	// Ставим статус "Не оплачен"
	Попытка Док.Статус = Перечисления.СтатусыОплатыСчетов.НеОплачен; Исключение КонецПопытки;

	Попытка
		Док.Записать(РежимЗаписиДокумента.Проведение);
	Исключение
		Док.Записать(РежимЗаписиДокумента.Запись);
	КонецПопытки;

	Возврат Док.Ссылка;
КонецФункции

Функция СоздатьПоступлениеНаСчет(ДанныеJson) Экспорт
	
	Организация = ПолучитьОрганизациюПоУмолчанию();
	Контрагент  = НайтиИлиСоздатьКонтрагента(ДанныеJson.client_name);
	Договор     = НайтиИлиСоздатьДоговор(Контрагент);
	
	Док = Документы.ПоступлениеНаРасчетныйСчет.СоздатьДокумент();
	
	// --- ШАГ 1: Инициализация (устанавливает Курс = 1 и Кратность = 1 автоматически) ---
	Док.Заполнить(Организация); 
	
	Док.Дата = ТекущаяДата();
	Док.Организация = Организация;
	Док.Контрагент  = Контрагент;
	Док.ВидОперации = Перечисления.ВидыОперацийПоступлениеДенежныхСредств.ОплатаПокупателя;
	
	// Валюта (643 - Рубли)
	ВалютаРуб = Справочники.Валюты.НайтиПоКоду("643");
	Док.ВалютаДокумента = ВалютаРуб;

	// Банковские счета ИП
	Док.СчетОрганизации = Организация.ОсновнойБанковскийСчет; 
	Док.СчетБанк = ПланыСчетов.Хозрасчетный.НайтиПоКоду("51");
	
	// Поиск Счета для связи
	Запрос = Новый Запрос("ВЫБРАТЬ ПЕРВЫЕ 1 B24_ВходящийПакет.СозданныйДокумент КАК Счет ИЗ Документ.B24_ВходящийПакет КАК B24_ВходящийПакет ГДЕ B24_ВходящийПакет.BitrixID = &BitrixID И B24_ВходящийПакет.ТипСобытия = ""invoice"" УПОРЯДОЧИТЬ ПО B24_ВходящийПакет.Дата УБЫВ");
	Запрос.УстановитьПараметр("BitrixID", ДанныеJson.deal_id);
	Выборка = Запрос.Выполнить().Выбрать();
	СчетСсылка = Неопределено;
	Если Выборка.Следующий() Тогда 
		СчетСсылка = Выборка.Счет;
		Попытка Док.СчетНаОплату = СчетСсылка; Исключение КонецПопытки;
	КонецЕсли;

	// Расчет суммы (безопасный)
	ОбщаяСумма = 0;
	Для Каждого ТекТовар Из ДанныеJson.products Цикл
		ОбщаяСумма = ОбщаяСумма + (Число(ТекТовар.quantity) * Число(ТекТовар.price));
	КонецЦикла;
	Док.СуммаДокумента = ОбщаяСумма;

	// --- ШАГ 2: ЗАПОЛНЕНИЕ ТАБЛИЦЫ ---
	Док.РасшифровкаПлатежа.Очистить(); // Очищаем то, что наполнил метод Заполнить()
	НовСтр = Док.РасшифровкаПлатежа.Добавить();
	
	НовСтр.СуммаПлатежа = ОбщаяСумма;
	НовСтр.ДоговорКонтрагента = Договор;
	НовСтр.СчетНаОплату = СчетСсылка; 

	// ПРИНУДИТЕЛЬНОЕ УСТРАНЕНИЕ ДЕЛЕНИЯ НА 0
	НовСтр.КурсВзаиморасчетов = 1;
	НовСтр.КратностьВзаиморасчетов = 1;
	НовСтр.СуммаВзаиморасчетов = ОбщаяСумма;
	
	НовСтр.СчетУчетаРасчетовСКонтрагентом = ПланыСчетов.Хозрасчетный.НайтиПоКоду("62.01");
	НовСтр.СчетУчетаРасчетовПоАвансам      = ПланыСчетов.Хозрасчетный.НайтиПоКоду("62.02");
	
	НовСтр.СтавкаНДС = Перечисления.СтавкиНДС.НДС20;
	НовСтр.СуммаНДС  = Окр(ОбщаяСумма * 20 / 120, 2);
	
	НовСтр.СпособПогашенияЗадолженности = Перечисления.СпособыПогашенияЗадолженности.Автоматически;

	// Статья ДДС
	Статья = Справочники.СтатьиДвиженияДенежныхСредств.НайтиПоНаименованию("Оплата от покупателей");
	Если Статья.Пустая() Тогда
		ВыбС = Справочники.СтатьиДвиженияДенежныхСредств.Выбрать();
		Если ВыбС.Следующий() Тогда Статья = ВыбС.Ссылка; КонецЕсли;
	КонецЕсли;
	НовСтр.СтатьяДвиженияДенежныхСредств = Статья;

	Док.НазначениеПлатежа = "Автоматическая оплата сделки №" + ДанныеJson.deal_id;

	// --- ШАГ 3: ЗАПИСЬ ---
	Попытка
		Док.Записать(РежимЗаписиДокумента.Проведение);
		// Письмо
		//Если ЗначениеЗаполнено(ДанныеJson.client_email) Тогда
		//	ОтправитьДокументНаПочту(Док.Ссылка, ДанныеJson.client_email);
		//КонецЕсли;
	Исключение
		// Если проведение не удалось, просто сохраняем
		Док.Записать(РежимЗаписиДокумента.Запись);
		ЗаписьЖурналаРегистрации("B24_Payment_Error", УровеньЖурналаРегистрации.Ошибка, , , ОписаниеОшибки());
	КонецПопытки;

	Возврат Док.Ссылка;
КонецФункции

Функция ПолучитьДанныеСделкиИзБ24(DealID)
	Результат = Новый Структура("deal_id, client_name, client_email, products", DealID, "Покупатель Б24", "", Новый Массив);
	
	Парам = Новый Соответствие; 
	Парам.Вставить("id", DealID);
	
	// 1. Получаем данные самой сделки
	ОтветСделки = ВыполнитьЗапросКБитрикс("crm.deal.get", Парам);
	
	Если ОтветСделки <> Неопределено И ОтветСделки.Свойство("result") Тогда
		 ДанныеСделки = ОтветСделки.result;
		 Результат.client_name = ДанныеСделки.TITLE;
		 
		IDКомпании = ?(ДанныеСделки.Свойство("COMPANY_ID"), Строка(ДанныеСделки.COMPANY_ID), "0");
		IDКонтакта = ?(ДанныеСделки.Свойство("CONTACT_ID"), Строка(ДанныеСделки.CONTACT_ID), "0");
		
		 Если ЗначениеЗаполнено(IDКомпании) И IDКомпании <> "0" Тогда
			ПарамК = Новый Соответствие; ПарамК.Вставить("id", IDКомпании);
			ОтветК = ВыполнитьЗапросКБитрикс("crm.company.get", ПарамК);
			Если ОтветК <> Неопределено И ОтветК.Свойство("result") Тогда
				Результат.client_name  = ОтветК.result.TITLE;
				Результат.client_email = ПолучитьEmailИзБ24(ОтветК.result);
			КонецЕсли;
			
		// 2. Если компании нет, ищем Контакт
		ИначеЕсли ЗначениеЗаполнено(IDКонтакта) И IDКонтакта <> "0" Тогда
			ПарамК = Новый Соответствие; ПарамК.Вставить("id", IDКонтакта);
			ОтветК = ВыполнитьЗапросКБитрикс("crm.contact.get", ПарамК);
			Если ОтветК <> Неопределено И ОтветК.Свойство("result") Тогда
				// Собираем ФИО
				Имя = ?(ОтветК.result.Свойство("NAME"), ОтветК.result.NAME, "");
				Фамилия = ?(ОтветК.result.Свойство("LAST_NAME"), ОтветК.result.LAST_NAME, "");
				Результат.client_name  = СокрЛП(Имя + " " + Фамилия);
				Результат.client_email = ПолучитьEmailИзБ24(ОтветК.result);
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	// Товары (без изменений)
	ОтветТоваров = ВыполнитьЗапросКБитрикс("crm.deal.productrows.get", Парам);
	Если ОтветТоваров <> Неопределено И ОтветТоваров.Свойство("result") Тогда
		Для Каждого ТекСтр Из ОтветТоваров.result Цикл
			Товар = Новый Структура("name, price, quantity", ТекСтр.PRODUCT_NAME, Число(ТекСтр.PRICE), Число(ТекСтр.QUANTITY));
			Результат.products.Добавить(Товар);
		КонецЦикла;
	КонецЕсли;
	
	Возврат Результат;
КонецФункции

	Запрос = Новый HTTPЗапрос(URL + СтрокаПарам);
	Попытка
		Ответ = Соединение.Получить(Запрос);

			Возврат ПрочитатьJSON(Чтение, Ложь);

Функция НайтиИлиСоздатьНоменклатуру(Имя)
	Н = Справочники.Номенклатура.НайтиПоНаименованию(Имя);
	
	Если Н.Пустая() Тогда
		Э = Справочники.Номенклатура.СоздатьЭлемент();
		Э.Наименование = Имя; 
		Э.НаименованиеПолное = Имя;
		
		// 1. Устанавливаем вид номенклатуры (Товары)

Функция СоздатьРеализацию(ДанныеJson) Экспорт
    
    Организация = ПолучитьОрганизациюПоУмолчанию();
    Контрагент  = НайтиИлиСоздатьКонтрагента(ДанныеJson.client_name);
    Договор     = НайтиИлиСоздатьДоговор(Контрагент);
    
    Склад = Справочники.Склады.НайтиПоНаименованию("Основной склад");
    Если Склад.Пустая() Тогда Склад = ПолучитьПервуюСсылку(Справочники.Склады); КонецЕсли;

    Док = Документы.РеализацияТоваровУслуг.СоздатьДокумент();
    Док.Заполнить(Организация);
    
    Док.Дата = ТекущаяДата();
    Док.Организация = Организация;
    Док.Контрагент = Контрагент;
    Док.ДоговорКонтрагента = Договор;
    Док.Склад = Склад;
    Док.СуммаВключаетНДС = Истина;
    Док.СпособЗачетаАвансов = Перечисления.СпособыЗачетаАвансов.Автоматически;

    // --- ПОИСК СЧЕТА И ЗАПОЛНЕНИЕ ТОВАРОВ ---
    Запрос = Новый Запрос("ВЫБРАТЬ ПЕРВЫЕ 1 B24_ВходящийПакет.СозданныйДокумент КАК Счет ИЗ Документ.B24_ВходящийПакет КАК B24_ВходящийПакет ГДЕ B24_ВходящийПакет.BitrixID = &BitrixID И B24_ВходящийПакет.ТипСобытия = ""invoice"" УПОРЯДОЧИТЬ ПО B24_ВходящийПакет.Дата УБЫВ");
    Запрос.УстановитьПараметр("BitrixID", ДанныеJson.deal_id);
    Выборка = Запрос.Выполнить().Выбрать();
    
    СчетЗаполнен = Ложь;
    Если Выборка.Следующий() И ЗначениеЗаполнено(Выборка.Счет) Тогда
        Док.СчетНаОплатуПокупателю = Выборка.Счет;
        ОбъектСчета = Выборка.Счет.ПолучитьОбъект();
        
        // Если в счете ЕСТЬ товары, копируем
        Если ОбъектСчета.Товары.Количество() > 0 Тогда
            Для Каждого СтрСчета Из ОбъектСчета.Товары Цикл
                НовСтр = Док.Товары.Добавить();
                ЗаполнитьЗначенияСвойств(НовСтр, СтрСчета);
                ЗаполнитьСчетаВСтроке(НовСтр, Док); 
            КонецЦикла;
            СчетЗаполнен = Истина;
        КонецЕсли;
    КонецЕсли;

    Если Не СчетЗаполнен Тогда
        Для Каждого ТекТовар Из ДанныеJson.products Цикл
            НовСтр = Док.Товары.Добавить();
            НовСтр.Номенклатура = НайтиИлиСоздатьНоменклатуру(ТекТовар.name);
            НовСтр.Количество   = Число(ТекТовар.quantity);
            НовСтр.Цена         = Число(ТекТовар.price);
            НовСтр.СтавкаНДС    = Перечисления.СтавкиНДС.НДС20;
            НовСтр.Сумма        = НовСтр.Количество * НовСтр.Цена;
            НовСтр.СуммаНДС     = Окр(НовСтр.Сумма * 20 / 120, 2);
            ЗаполнитьСчетаВСтроке(НовСтр, Док);
        КонецЦикла;
    КонецЕсли;

    Док.СуммаДокумента = Док.Товары.Итог("Сумма");

    Попытка
        Док.Записать(РежимЗаписиДокумента.Проведение);
    Исключение
        Док.Записать(РежимЗаписиДокумента.Запись);
    КонецПопытки;

    Возврат Док.Ссылка;
